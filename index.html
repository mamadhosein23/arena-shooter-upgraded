/** * Entity movement module with Continuous Collision Detection (CCD) substapping * and proper vector projection / axis sliding mechanics. * * @param {Object} entity - An object containing { x, y } * @param {number} dx - Displacement vector X (already scaled by delta) * @param {number} dy - Displacement vector Y (already scaled by delta) * @param {number} radius - Entity collision bound radius * @param {Function} wallCollide - Collision predicate: (x, y, radius) => boolean */ function moveEntity(entity, dx, dy, radius, wallCollide) { const distance = Math.hypot(dx, dy); if (distance === 0) return; // Sub-stepping to prevent tunneling at high velocity (limit step to radius / 2) const maxStepSize = Math.max(radius * 0.5, 1); const steps = Math.ceil(distance / maxStepSize); const stepX = dx / steps; const stepY = dy / steps; for (let i = 0; i < steps; i++) { const nextX = entity.x + stepX; const nextY = entity.y + stepY; // 1. Try full unconstrained step first if (!wallCollide(nextX, nextY, radius)) { entity.x = nextX; entity.y = nextY; continue; } // 2. Collision detected: Resolve sliding on isolated axes const canMoveX = !wallCollide(nextX, entity.y, radius); const canMoveY = !wallCollide(entity.x, nextY, radius); if (canMoveX) { entity.x = nextX; } if (canMoveY) { entity.y = nextY; } // If completely blocked on both axes, halt further substeps for this tick if (!canMoveX && !canMoveY) { break; } } } /** * High-performance Time-sliced A* scheduler. * Utilizes a Set for O(1) duplicate lookups and pointer-based queue consuming. */ class PathfindingScheduler { /** * @param {number} maxPathsPerFrame - Maximum pathfinding computations allowed per tick */ constructor(maxPathsPerFrame = 5) { this.maxPathsPerFrame = maxPathsPerFrame; this.queue = []; this.pendingEnemies = new Set(); } /** * Adds an entity to the pathfinding queue with O(1) deduplication. * * @param {Object} enemy - Target entity requiring path * @param {Object} target - Destination coordinate or entity * @param {Function} findPathCallback - Pure pathfinding execution function */ requestPath(enemy, target, findPathCallback) { if (this.pendingEnemies.has(enemy)) return; this.pendingEnemies.add(enemy); this.queue.push({ enemy, target, findPathCallback }); } /** * Processes queued path calculations within the frame budget. */ update() { if (this.queue.length === 0) return; const count = Math.min(this.queue.length, this.maxPathsPerFrame); for (let i = 0; i < count; i++) { const request = this.queue.shift(); // For extreme scale (>10k), swap with a ring buffer this.pendingEnemies.delete(request.enemy); // Null-check / active check to prevent computing paths for dead entities if (request.enemy.isDestroyed || !request.enemy.active) { continue; } request.enemy.path = request.findPathCallback(request.enemy, request.target); } } /** * Clears all pending jobs. Call on state resets or level unload. */ clear() { this.queue.length = 0; this.pendingEnemies.clear(); } } // ========================================== // Usage Example (FixedUpdate) // ========================================== const pathScheduler = new PathfindingScheduler(5); function fixedUpdate(delta) { // 1.

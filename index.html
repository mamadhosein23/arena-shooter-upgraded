/** * Entity movement module with Continuous Collision Detection (CCD) * and sliding mechanics to prevent tunneling. */ function moveEntity(entity, dx, dy, radius, wallCollide) { const distance = Math.hypot(dx, dy); if (distance === 0) return; // Calculate dynamic steps based on radius to eliminate tunneling at high speeds const maxStepSize = Math.max(radius * 0.5, 1); const steps = Math.ceil(distance / maxStepSize); const stepX = dx / steps; const stepY = dy / steps; for (let i = 0; i < steps; i++) { // Horizontal movement with wall sliding if (!wallCollide(entity.x + stepX, entity.y, radius)) { entity.x += stepX; } // Vertical movement if (!wallCollide(entity.x, entity.y + stepY, radius)) { entity.y += stepY; } } } /** * Time-sliced A* scheduler. * Limits CPU usage by distributing pathfinding requests over multiple frames. */ class PathfindingScheduler { constructor(maxPathsPerFrame = 5) { this.maxPathsPerFrame = maxPathsPerFrame; this.queue = []; } /** * Add a pathfinding request to the queue. */ requestPath(enemy, target, findPathCallback) { // Prevent duplicate requests for the same enemy if (!this.queue.some(req => req.enemy === enemy)) { this.queue.push({ enemy, target, findPathCallback }); } } /** * Processes pending requests in the game loop. */ update() { let processed = 0; while (this.queue.length > 0 && processed < this.maxPathsPerFrame) { const { enemy, target, findPathCallback } = this.queue.shift(); // Only update path if necessary (e.g., target moved, LOS broken) enemy.path = findPathCallback(enemy, target); processed++; } } } // ========================================== // Usage Example (FixedUpdate) // ========================================== const pathScheduler = new PathfindingScheduler(5); function fixedUpdate(delta) { // 1.

/** * اصلاحیه: مدیریت دقیق CCD و Sliding * بهبود: استفاده از Cross Product برای مدیریت گوشه‌ها (Corner Sliding) */ function moveEntityCCD(entity, dx, dy, radius, castCircle, opts = {}) { // ... (اعتبارسنجی ورودی‌ها همانند قبل باقی می‌ماند) const maxIters = Math.max(1, Math.floor(opts.maxIters ?? 6)); const skin = Math.max(0, opts.skin ?? 0.005); const minMoveSq = (opts.minMove ?? 1e-6) ** 2; const friction = clamp01(opts.friction ?? 0); const restitution = clamp01(opts.restitution ?? 0); let remX = dx, remY = dy; let normalCount = 0; const normals = opts.normalsBuffer ?? new Float64Array(maxIters * 2); for (let iter = 0; iter < maxIters; iter++) { const distSq = remX * remX + remY * remY; if (distSq <= minMoveSq) break; const hit = castCircle(entity.x, entity.y, radius, remX, remY); if (!hit) { entity.x += remX; entity.y += remY; break; } // 1. پیشروی تا نزدیک سطح (با لحاظ Skin) const t = Math.max(0, Math.min(1, hit.t)); const dist = Math.sqrt(distSq); // اصلاح دقیق Skin: حرکت تا t درصد مسیر، منهای مقدار skin برای جلوگیری از چسبیدن const moveDist = dist * t; const actualMoveDist = Math.max(0, moveDist - skin); const fraction = (dist > 0) ? (actualMoveDist / dist) : 0; entity.x += remX * fraction; entity.y += remY * fraction; // به‌روزرسانی بردار باقی‌مانده پس از برخورد const nx = hit.nx, ny = hit.ny; // فرض بر نرمالیزه بودن از سمت تابع cast // 2. مدیریت جمع‌آوری نرمال‌ها (جلوگیری از نرمال‌های تکراری در سطوح تخت) let duplicate = false; for (let i = 0; i < normalCount; i++) { if (normals[i * 2] * nx + normals[i * 2 + 1] * ny > 0.999) { duplicate = true; break; } } if (!duplicate) { normals[normalCount * 2] = nx; normals[normalCount * 2 + 1] = ny; normalCount++; } // 3. محاسبه بردار لغزش (Sliding Plane) // فرمول کلاسیک: Velocity = Velocity - (Velocity ⋅ Normal) * Normal const dot = remX * nx + remY * ny; remX -= dot * nx; remY -= dot * ny; // اعمال اصطکاک روی بردار مماسی remX *= (1 - friction); remY *= (1 - friction); // 4. جلوگیری از نفوذ مجدد به سطوح (تکرار برای هندسه‌های پیچیده/گوشه‌ها) // اگر بعد از کسر نرمال، هنوز بردار حرکت به سمت داخل نرمال قبلی باشد، آن را صفر می‌کنیم. for (let i = 0; i < normalCount; i++) { const dotNew = remX * normals[i * 2] + remY * normals[i * 2 + 1]; if (dotNew < 0) { remX -= dotNew * normals[i * 2]; remY -= dotNew * normals[i * 2 + 1]; } } } return { x: entity.x, y: entity.y }; }

/** * Robust 2D Kinematic Swept-Circle Move & Slide * * castCircle(x, y, radius, dx, dy) must return: * null * or { * t: number, // normalized TOI in [0, 1] * nx: number, // collision normal, pointing out of obstacle * ny: number, * startedInside?: boolean, * depth?: number // penetration depth when startedInside is true * } */ function moveEntityCCD(entity, dx, dy, radius, castCircle, opts = {}) { validateInputs(entity, dx, dy, radius, castCircle); const maxIters = clampInt( finiteOption(opts.maxIters, 6, "maxIters"), 1, 32 ); const skin = Math.max( 0, finiteOption(opts.skin, 0.005, "skin") ); const minMove = Math.max( 0, finiteOption(opts.minMove, 1e-6, "minMove") ); const normalEpsilon = Math.max( 0, finiteOption(opts.normalEpsilon, 1e-8, "normalEpsilon") ); const duplicateDot = clamp( finiteOption(opts.duplicateDot, 0.9995, "duplicateDot"), -1, 1 ); const friction = clamp01( finiteOption(opts.friction, 0, "friction") ); const restitution = clamp01( finiteOption(opts.restitution, 0, "restitution") ); const collectHits = opts.collectHits === true; const startX = entity.x; const startY = entity.y; let remX = dx; let remY = dy; let hitCount = 0; let iterations = 0; let blocked = false; let startedInside = false; const collisions = collectHits ? [] : null; const planesX = new Float64Array(maxIters); const planesY = new Float64Array(maxIters); let planeCount = 0; const minMoveSq = minMove * minMove; for (let iter = 0; iter < maxIters; iter++) { const remLengthSq = remX * remX + remY * remY; if (remLengthSq <= minMoveSq) { remX = 0; remY = 0; break; } iterations = iter + 1; const hit = castCircle( entity.x, entity.y, radius, remX, remY ); if (!hit) { entity.x += remX; entity.y += remY; remX = 0; remY = 0; break; } validateHit(hit); const normalLength = Math.hypot(hit.nx, hit.ny); if (normalLength <= 1e-12) { throw new TypeError( "castCircle collision normal must have non-zero length" ); } const nx = hit.nx / normalLength; const ny = hit.ny / normalLength; const t = clamp(hit.t, 0, 1); /* * Initial-overlap handling * * A sweep alone cannot reliably resolve an existing penetration. * If castCircle provides penetration depth, depenetrate first. */ if (hit.startedInside === true) { startedInside = true; if (!Number.isFinite(hit.depth) || hit.depth < 0) { blocked = true; break; } const pushDistance = hit.depth + skin; entity.x += nx * pushDistance; entity.y += ny * pushDistance; if (collectHits) { collisions.push({ t: 0, nx, ny, startedInside: true, depth: hit.depth, }); } hitCount++; continue; } hitCount++; if (collectHits) { collisions.push({ t, nx, ny, startedInside: false, }); } const remLength = Math.sqrt(remLengthSq); /* * Move to just before contact. * * Important: * skin affects only the position advance. It must not give the entity * additional simulation time. */ const skinFraction = remLength > 1e-12 ? skin / remLength : 0; const advanceFraction = clamp(t - skinFraction, 0, 1); entity.x += remX * advanceFraction; entity.y += remY * advanceFraction; /* * Remaining displacement is based on the actual TOI, not on the * skin-adjusted advance fraction. * * Using (1 - advanceFraction) here would create extra movement/time. */ const remainingFraction = 1 - t; remX *= remainingFraction; remY *= remainingFraction; // Register a unique contact plane. let duplicatePlane = false; for (let i = 0; i < planeCount; i++) { const planeDot = planesX[i] * nx + planesY[i] * ny; if (planeDot >= duplicateDot) { duplicatePlane = true; break; } } if (!duplicatePlane && planeCount < maxIters) { planesX[planeCount] = nx; planesY[planeCount] = ny; planeCount++; } /* * Apply response against the newly hit surface. * * Friction is applied only to the tangent of the new contact. * Old planes are used only as non-penetration constraints. */ const intoSurface = remX * nx + remY * ny; if (intoSurface < -normalEpsilon) { const normalX = nx * intoSurface; const normalY = ny * intoSurface; const tangentScale = 1 - friction; const tangentX = (remX - normalX) * tangentScale; const tangentY = (remY - normalY) * tangentScale; remX = tangentX - normalX * restitution; remY = tangentY - normalY * restitution; } /* * Project the candidate displacement into the feasible region defined * by all accumulated contact planes. * * Repeated projection is appropriate here because this is 2D and the * number of contact planes is small. */ const maxProjectionPasses = Math.max(1, planeCount); for (let pass = 0; pass < maxProjectionPasses; pass++) { let changed = false; for (let i = 0; i < planeCount; i++) { const px = planesX[i]; const py = planesY[i]; const intoPlane = remX * px + remY * py; if (intoPlane < -normalEpsilon) { remX -= px * intoPlane; remY -= py * intoPlane; changed = true; } } if (!changed) { break; } } /* * Verify that projection satisfies every contact constraint. * If not, the entity is trapped by incompatible planes. */ let violatesPlane = false; for (let i = 0; i < planeCount; i++) { const dot = remX * planesX[i] + remY * planesY[i]; if (dot < -normalEpsilon) { violatesPlane = true; break; } } if (violatesPlane) { remX = 0; remY = 0; blocked = true; break; } if (remX * remX + remY * remY <= minMoveSq) { remX = 0; remY = 0; break; } /* * A valid shape cast should not repeatedly return t=0 for tangential * or separating motion. Stop safely if it does, preventing an infinite * contact loop. */ if ( t <= 1e-12 && remX * nx + remY * ny >= -normalEpsilon ) { blocked = true; break; } } const remainingLengthSq = remX * remX + remY * remY; const result = { x: entity.x, y: entity.y, movedX: entity.x - startX, movedY: entity.y - startY, remainingX: remX, remainingY: remY, hitCount, iterations, blocked, startedInside, truncated: remainingLengthSq > minMoveSq && iterations >= maxIters, }; if (collectHits) { result.collisions = collisions; } return result; } function validateInputs(entity, dx, dy, radius, castCircle) { if ( !entity || !Number.isFinite(entity.x) || !Number.isFinite(entity.y) ) { throw new TypeError("entity must have finite x and y"); } if (!Number.isFinite(dx) || !Number.isFinite(dy)) { throw new TypeError("dx and dy must be finite"); } if (!Number.isFinite(radius) || radius < 0) { throw new RangeError( "radius must be a finite non-negative number" ); } if (typeof castCircle !== "function") { throw new TypeError("castCircle must be a function"); } } function validateHit(hit) { if ( !hit || !Number.isFinite(hit.t) || !Number.isFinite(hit.nx) || !Number.isFinite(hit.ny) ) { throw new TypeError( "castCircle must return null or a valid hit object" ); } } function finiteOption(value, fallback, name) { if (value === undefined) { return fallback; } if (!Number.isFinite(value)) { throw new TypeError(`${name} must be finite`); } return value; } function clampInt(value, min, max) { return Math.min(max, Math.max(min, Math.floor(value))); } function clamp01(value) { return clamp(value, 0, 1); } function clamp(value, min, max) { return Math.min(max, Math.max(min, value)); }

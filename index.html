/** * Kinematic CCD Swept-Circle Move & Slide * Architecture: Valve Source Engine (ClipVelocity) & Quake III (PM_SlideMove) */ function moveEntityCCD(entity, dx, dy, radius, castCircle, opts = {}) { // ۱. اعتبارسنجی ورودی‌ها if (!entity || !Number.isFinite(entity.x) || !Number.isFinite(entity.y)) { throw new TypeError("entity must have finite x and y"); } if (!Number.isFinite(dx) || !Number.isFinite(dy)) { throw new TypeError("dx and dy must be finite"); } if (!Number.isFinite(radius) || radius < 0) { throw new RangeError("radius must be a finite non-negative number"); } if (typeof castCircle !== "function") { throw new TypeError("castCircle must be a function"); } // ۲. تنظیم پارامترها و ثابت‌ها const maxIters = Math.max(1, Math.floor(opts.maxIters ?? 5)); const skin = Math.max(0, opts.skin ?? 0.005); const minMove = Math.max(0, opts.minMove ?? 1e-6); const minMoveSq = minMove * minMove; const friction = clamp01(opts.friction ?? 0); const restitution = clamp01(opts.restitution ?? 0); const collectHits = opts.collectHits === true; const startX = entity.x; const startY = entity.y; const hits = collectHits ? [] : null; let remX = dx; let remY = dy; let hitCount = 0; let iterations = 0; // بافر اختصاصی نرمال‌ها بدون ایجاد Garbage Collection const normalsX = new Float64Array(maxIters); const normalsY = new Float64Array(maxIters); let numNormals = 0; // ۳. حلقه شبیه‌سازی حرکت for (let iter = 0; iter < maxIters; iter++) { iterations = iter + 1; const distSq = remX * remX + remY * remY; if (distSq <= minMoveSq) { remX = 0; remY = 0; break; } const hit = castCircle(entity.x, entity.y, radius, remX, remY); // حرکت آزاد بدون برخورد if (!hit) { entity.x += remX; entity.y += remY; remX = 0; remY = 0; break; } // اعتبارسنجی خروجی برخورد if ( !Number.isFinite(hit.t) || !Number.isFinite(hit.nx) || !Number.isFinite(hit.ny) ) { throw new TypeError("castCircle returned an invalid hit"); } const nLen = Math.hypot(hit.nx, hit.ny); if (nLen < 1e-12) { throw new TypeError("collision normal must have non-zero length"); } const nx = hit.nx / nLen; const ny = hit.ny / nLen; const t = Math.max(0, Math.min(1, hit.t)); hitCount++; if (collectHits) { hits.push({ t, nx, ny }); } // ۴. پیشروی ایمن تا نزدیکی سطح (محاسبه دقیق فاصله با احتساب skin) const dist = Math.sqrt(distSq); const safeDist = Math.max(0, dist * t - skin); const fraction = dist > 1e-12 ? safeDist / dist : 0; entity.x += remX * fraction; entity.y += remY * fraction; // ۵. اصلاح بردار حرکت باقی‌مانده متناسب با زمان صرف‌شده (جلوگیری از تولید انرژی کاذب) const timeRemaining = 1 - fraction; remX *= timeRemaining; remY *= timeRemaining; // ثبت نرمال‌های غیرتکراری سطح let isDuplicate = false; for (let i = 0; i < numNormals; i++) { if (normalsX[i] * nx + normalsY[i] * ny > 0.999) { isDuplicate = true; break; } } if (!isDuplicate && numNormals < maxIters) { normalsX[numNormals] = nx; normalsY[numNormals] = ny; numNormals++; } // ۶. برش بردار حرکت (ClipVelocity با در نظر گرفتن Bounce و Friction) const into = remX * nx + remY * ny; if (into < 0) { const normalCompX = into * nx; const normalCompY = into * ny; // اعمال اصطکاک روی مؤلفه مماس و جهش روی مؤلفه عمود const tangX = (remX - normalCompX) * (1 - friction); const tangY = (remY - normalCompY) * (1 - friction); remX = tangX - normalCompX * restitution; remY = tangY - normalCompY * restitution; } // ۷. بررسی گیر افتادن در کنج و گوشه‌ها (Crease / Corner Handling) // در فضای ۲بعدی، اگر بردار جدید به سمت داخل هر یک از سطوح تماس قبلی باشد، گیر کرده است let trapped = false; for (let i = 0; i < numNormals; i++) { if (remX * normalsX[i] + remY * normalsY[i] < -minMove) { trapped = true; break; } } if (trapped) { remX = 0; remY = 0; break; } } const result = { x: entity.x, y: entity.y, movedX: entity.x - startX, movedY: entity.y - startY, hitCount, iterations, remainingX: remX, remainingY: remY, truncated: remX * remX + remY * remY > minMoveSq, }; if (collectHits) result.collisions = hits; return result; } function clamp01(value) { return value < 0 ? 0 : value > 1 ? 1 : value; }

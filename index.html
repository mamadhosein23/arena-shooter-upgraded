/** * Move & slide a circular entity using swept-circle CCD. * Reference: Valve Source Engine `ClipVelocity` & Quake-style kinematic sweep/slide. */ function moveEntityCCD(entity, dx, dy, radius, castCircle, opts = {}) { if (!entity || !Number.isFinite(entity.x) || !Number.isFinite(entity.y)) { throw new TypeError("entity must have finite x and y"); } if (!Number.isFinite(dx) || !Number.isFinite(dy)) { throw new TypeError("dx and dy must be finite"); } if (!Number.isFinite(radius) || radius < 0) { throw new RangeError("radius must be a finite non-negative number"); } if (typeof castCircle !== "function") { throw new TypeError("castCircle must be a function"); } const maxIters = Math.max(1, Math.floor(opts.maxIters ?? 6)); const skin = Math.max(0, opts.skin ?? 0.005); const minMove = Math.max(0, opts.minMove ?? 1e-6); const minMoveSq = minMove * minMove; const friction = clamp01(opts.friction ?? 0); const restitution = clamp01(opts.restitution ?? 0); const collectHits = opts.collectHits === true; const normals = opts.normalsBuffer ?? new Float64Array(maxIters * 2); if (!(normals instanceof Float64Array) || normals.length < maxIters * 2) { throw new RangeError("normalsBuffer must be a Float64Array of length >= maxIters * 2"); } const startX = entity.x; const startY = entity.y; const hits = collectHits ? [] : null; let remX = dx; let remY = dy; let normalCount = 0; let hitCount = 0; let iterations = 0; for (let iter = 0; iter < maxIters; iter++) { iterations = iter + 1; const distSq = remX * remX + remY * remY; if (distSq <= minMoveSq) { remX = 0; remY = 0; break; } const dist = Math.sqrt(distSq); const hit = castCircle(entity.x, entity.y, radius, remX, remY); if (!hit) { entity.x += remX; entity.y += remY; remX = 0; remY = 0; break; } if ( !Number.isFinite(hit.t) || !Number.isFinite(hit.nx) || !Number.isFinite(hit.ny) ) { throw new TypeError("castCircle returned an invalid hit"); } const t = Math.max(0, Math.min(1, hit.t)); const normalLength = Math.hypot(hit.nx, hit.ny); if (normalLength < 1e-12) { throw new TypeError("collision normal must have non-zero length"); } const nx = hit.nx / normalLength; const ny = hit.ny / normalLength; hitCount++; // اصلاح ۱: پیشروی تا قبل از برخورد بر اساس skin const skinRatio = skin / dist; const safeT = Math.max(0, Math.min(t, t - skinRatio)); entity.x += remX * safeT; entity.y += remY * safeT; // اصلاح ۲: همگام‌سازی جابه‌جایی باقیمانده با مسافت طی‌شده const remainingFraction = 1 - safeT; let nextX = remX * remainingFraction; let nextY = remY * remainingFraction; // افزودن نرمال با فیلتر زوایای بسیار نزدیک let duplicate = false; for (let i = 0; i < normalCount; i++) { const px = normals[i * 2]; const py = normals[i * 2 + 1]; if (px * nx + py * ny > 0.999) { duplicate = true; break; } } if (!duplicate && normalCount < maxIters) { normals[normalCount * 2] = nx; normals[normalCount * 2 + 1] = ny; normalCount++; } // اعمال اصطکاک مماسی و ضریب بازگشت (Bounce) const normalSpeed = nextX * nx + nextY * ny; if (normalSpeed < 0) { const tangentX = nextX - normalSpeed * nx; const tangentY = nextY - normalSpeed * ny; nextX = tangentX * (1 - friction) - normalSpeed * restitution * nx; nextY = tangentY * (1 - friction) - normalSpeed * restitution * ny; } // حل تداخل چندوجهی (Projecting onto feasible sliding cone) const projected = projectToContactCone( nextX, nextY, normals, normalCount, minMove ); remX = projected[0]; remY = projected[1]; if (collectHits) { hits.push({ t, nx, ny }); } // توقف در صورت صفر شدن بردار حرکت ناشی از گیر افتادن در گوشه if (remX * remX + remY * remY <= minMoveSq) { remX = 0; remY = 0; break; } } const result = { x: entity.x, y: entity.y, movedX: entity.x - startX, movedY: entity.y - startY, hitCount, iterations, remainingX: remX, remainingY: remY, truncated: remX * remX + remY * remY > minMoveSq, }; if (collectHits) result.collisions = hits; return result; } function projectToContactCone(vx, vy, normals, count, epsilon) { let feasible = true; for (let i = 0; i < count; i++) { const nx = normals[i * 2]; const ny = normals[i * 2 + 1]; if (vx * nx + vy * ny < -epsilon) { feasible = false; break; } } if (feasible) return [vx, vy]; let bestX = 0; let bestY = 0; let bestDistSq = 0; // به دنبال بیشترین طول حرکت مجاز در راستای درست for (let i = 0; i < count; i++) { const nx = normals[i * 2]; const ny = normals[i * 2 + 1]; const into = vx * nx + vy * ny; // تصویر روی سطح تماس i const candidateX = vx - into * nx; const candidateY = vy - into * ny; let candidateFeasible = true; for (let j = 0; j < count; j++) { if (i === j) continue; const jx = normals[j * 2]; const jy = normals[j * 2 + 1]; if (candidateX * jx + candidateY * jy < -epsilon) { candidateFeasible = false; break; } } if (candidateFeasible) { const magSq = candidateX * candidateX + candidateY * candidateY; if (magSq > bestDistSq) { bestDistSq = magSq; bestX = candidateX; bestY = candidateY; } } } // اصلاح ۳: بررسی تقاطع گوشه‌ها (V-shape Corner / Crease Slide) // در صورتی که هیچ خط مماسی منفردی مجاز نبود، بردار حرکت در راستای تلاقی دو صفحه (در دوبعدی: 0) است return [bestX, bestY]; } function clamp01(value) { return Math.max(0, Math.min(1, value)); }

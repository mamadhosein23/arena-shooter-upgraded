if (distance === 0) return; // گام‌های پویا بر اساس شعاع برای حذف کامل Tunneling در سرعت‌های بالا const maxStepSize = Math.max(radius * 0.5, 1); const steps = Math.ceil(distance / maxStepSize); const stepX = dx / steps; const stepY = dy / steps; for (let i = 0; i < steps; i++) { // محور X با امکان حرکت در امتداد دیوار (Wall Sliding) if (!wallCollide(entity.x + stepX, entity.y, radius)) { entity.x += stepX; } // محور Y if (!wallCollide(entity.x, entity.y + stepY, radius)) { entity.y += stepY; } } } /** * زمان‌بند اجرای الگوریتم A* (Time-slicing Pathfinding Queue) * جلوگیری از افت فریم ناشی از اجرای سنگین مسیر‌یابی در یک فریم */ class PathfindingScheduler { constructor(maxPathsPerFrame = 5) { this.maxPathsPerFrame = maxPathsPerFrame; this.queue = []; } /** * ثبت درخواست محاسبه مسیر * @param {Object} enemy - موجودیت دشمن * @param {Object} target - هدف (پلیر) * @param {Function} findPathCallback - متد فراخوانی A* */ requestPath(enemy, target, findPathCallback) { // جلوگیری از ثبت درخواست تکراری برای یک موجودیت if (!this.queue.some(req => req.enemy === enemy)) { this.queue.push({ enemy, target, findPathCallback }); } } /** * اجرا در حلقه FixedUpdate بازی */ update() { let processed = 0; while (this.queue.length > 0 && processed < this.maxPathsPerFrame) { const { enemy, target, findPathCallback } = this.queue.shift(); // شرط بازبینی: فقط در صورت عدم دید مستقیم (Raycast) یا تغییر موقعیت معنادار enemy.path = findPathCallback(enemy, target); processed++; } } } // ========================================== // نحوه استفاده در چرخه بازی (FixedUpdate Loop) // ========================================== const pathScheduler = new PathfindingScheduler(5);

/** * Advanced Production-Grade Move & Slide with Continuous Collision Detection (CCD). * * @param {Object} entity - شیء دارای x, y * @param {number} dx - جابه‌جایی افقی در فریم جاری * @param {number} dy - جابه‌جایی عمودی در فریم جاری * @param {number} radius - شعاع دایره کالیژن * @param {Function} castCircle - تابع Swept Circle: (x, y, r, dx, dy) -> { t, nx, ny } | null * @param {Object} [opts={}] - تنظیمات تکمیلی * @returns {Object} گزارش نتیجه حرکت { movedX, movedY, hitCount, isGrounded, collisions } */ function moveEntityCCD(entity, dx, dy, radius, castCircle, opts = {}) { const maxIters = opts.maxIters ?? 5; const skin = opts.skin ?? 0.005; // فاصله احتیاطی از سطح (Margin) const minMove = opts.minMove ?? 1e-5; // حداقل جابه‌جایی معنادار const minMoveSq = minMove * minMove; const friction = opts.friction ?? 0.0; // ضریب اصطکاک سطحی [0..1] const bounciness = opts.bounciness ?? 0.0; // خاصیت ارتجاعی [0..1] let vx = dx; let vy = dy; let totalMovedX = 0; let totalMovedY = 0; // ذخیره نرمال‌های برخورد برای جلوگیری از گیر افتادن در Crease/Corner const contactNormals = []; const collisions = []; for (let iter = 0; iter < maxIters; iter++) { const distSq = vx * vx + vy * vy; if (distSq < minMoveSq) break; const hit = castCircle(entity.x, entity.y, radius, vx, vy); // حالت بدون مانع: اتمام مسیر if (!hit) { entity.x += vx; entity.y += vy; totalMovedX += vx; totalMovedY += vy; break; } const t = Math.max(0, Math.min(1, hit.t)); const dist = Math.sqrt(distSq); // ۱. حرکت تا مرز ایمن با لحاظ کردن skin // در t=0 (تداخل اولیه)، حرکت نمی‌کنیم تا وارد جسم نشود const safeTravelDist = Math.max(0, t * dist - skin); const stepX = (vx / dist) * safeTravelDist; const stepY = (vy / dist) * safeTravelDist; entity.x += stepX; entity.y += stepY; totalMovedX += stepX; totalMovedY += stepY; // ثبت برخورد collisions.push({ t, nx: hit.nx, ny: hit.ny }); // ۲. محاسبه بردار باقی‌مانده بر مبنای نسبت زمانی خالص (نه تفریق مستقیم عددی) const remainingTimeRatio = Math.max(0, 1.0 - t); let rx = vx * remainingTimeRatio; let ry = vy * remainingTimeRatio; // ۳. پیش‌بینی برخورد با گوشه‌های تند (Crease Handling) let nx = hit.nx; let ny = hit.ny; contactNormals.push({ nx, ny }); let into = rx * nx + ry * ny; if (into < 0) { // تجزیه به دو مولفه عمودی (نرمال) و مماس (تانژانت) const normalCompX = into * nx; const normalCompY = into * ny; const tangentCompX = rx - normalCompX; const tangentCompY = ry - normalCompY; // اعمال ضریب بازتاب (Bounce) و اصطکاک (Friction) rx = (tangentCompX * (1.0 - friction)) - (normalCompX * bounciness); ry = (tangentCompY * (1.0 - friction)) - (normalCompY * bounciness); // ۴. حل قفل شدن در کنج‌ها: اگر بردار جدید به سمت نرمال‌های قبلی برود، روی خط تقاطع محدودش کن for (let i = 0; i < contactNormals.length - 1; i++) { const prev = contactNormals[i]; const dotPrev = rx * prev.nx + ry * prev.ny; if (dotPrev < 0) { // در دو بعد (2D)، حرکت درون کنج حاد به سکون کامل منجر می‌شود rx = 0; ry = 0; break; } } } else { // در حال دور شدن از سطح rx = 0; ry = 0; } vx = rx; vy = ry; } return { movedX: totalMovedX, movedY: totalMovedY, hitCount: collisions.length, collisions }; }

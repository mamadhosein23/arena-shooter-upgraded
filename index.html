/** * Move & Slide with TRUE CCD (time-of-impact) using swept circle casts. * * Required collision API: * castCircle(x, y, r, dx, dy) -> null | { t, nx, ny } * Where: * t in [0..1] is time of first impact along (dx, dy) * (nx, ny) is collision normal (unit length) */ function moveEntityCCD(entity, dx, dy, radius, castCircle, opts = {}) { const maxIters = opts.maxIters ?? 4; // max number of collisions per tick const skin = opts.skin ?? 0.01; // small margin to avoid jitter const minMove = opts.minMove ?? 1e-6; let vx = dx, vy = dy; for (let iter = 0; iter < maxIters; iter++) { const dist = Math.hypot(vx, vy); if (dist < minMove) break; // Cast circle along remaining movement const hit = castCircle(entity.x, entity.y, radius, vx, vy); // No collision: apply full movement if (!hit) { entity.x += vx; entity.y += vy; break; } // Move up to just before contact (leave skin margin) const t = Math.max(0, Math.min(1, hit.t)); const travel = Math.max(0, t * dist - skin); const tx = (vx / dist) * travel; const ty = (vy / dist) * travel; entity.x += tx; entity.y += ty; // Compute remaining motion after contact // remaining = original - traveled let rx = vx - tx; let ry = vy - ty; // Slide: remove the component into the normal const nx = hit.nx, ny = hit.ny; const into = rx * nx + ry * ny; // If moving into surface, project out that component if (into < 0) { rx = rx - into * nx; ry = ry - into * ny; } else { // Already moving away: stop to avoid numerical issues rx = 0; ry = 0; } vx = rx; vy = ry; } }

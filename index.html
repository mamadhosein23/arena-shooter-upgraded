/** * Kinematic CCD Swept-Circle Move & Slide * Standard Architecture: Quake III (PM_SlideMove) & Valve Source (ClipVelocity) * * @param {Object} entity - شیء شامل x و y * @param {number} vx - سرعت افقی (یا جابجایی اولیه dx به ازای فریم) * @param {number} vy - سرعت عمودی (یا جابجایی اولیه dy به ازای فریم) * @param {number} radius - شعاع دایره برخورد * @param {Function} castCircle - تابع Swept Circle: (x, y, r, dx, dy) => { t, nx, ny } | null * @param {Object} opts - تنظیمات فیزیکی شبیه‌سازی */ function moveEntityCCD(entity, vx, vy, radius, castCircle, opts = {}) { // ۱. اعتبارسنجی دقیق ورودی‌ها if (!entity || !Number.isFinite(entity.x) || !Number.isFinite(entity.y)) { throw new TypeError("entity must have finite x and y coordinates"); } if (!Number.isFinite(vx) || !Number.isFinite(vy)) { throw new TypeError("vx and vy must be finite"); } if (!Number.isFinite(radius) || radius < 0) { throw new RangeError("radius must be a finite non-negative number"); } if (typeof castCircle !== "function") { throw new TypeError("castCircle must be a function"); } // ۲. استخراج پارامترها و مقادیر آستانه const maxIters = Math.max(1, Math.floor(opts.maxIters ?? 4)); const skin = Math.max(0, opts.skin ?? 0.005); const minMove = Math.max(0, opts.minMove ?? 1e-5); const friction = clamp01(opts.friction ?? 0); const restitution = clamp01(opts.restitution ?? 0); const collectHits = opts.collectHits === true; const startX = entity.x; const startY = entity.y; const hits = collectHits ? [] : null; let curVx = vx; let curVy = vy; let timeLeft = 1.0; // کسری از کل زمان فریم باقی‌مانده (۰ تا ۱) let hitCount = 0; let iterations = 0; // بافر ایستا برای ثبت صفحات برخورد const planesX = new Float64Array(maxIters); const planesY = new Float64Array(maxIters); let numPlanes = 0; // ۳. حلقه شبیه‌سازی گام زمانی for (let iter = 0; iter < maxIters; iter++) { iterations = iter + 1; // محاسبه بردار جابجایی گام جاری let stepDx = curVx * timeLeft; let stepDy = curVy * timeLeft; const stepDistSq = stepDx * stepDx + stepDy * stepDy; // توقف در صورت ناچیز بودن انرژی حرکتی if (stepDistSq <= minMove * minMove) { break; } const hit = castCircle(entity.x, entity.y, radius, stepDx, stepDy); // الف) حرکت آزاد بدون مانع if (!hit) { entity.x += stepDx; entity.y += stepDy; timeLeft = 0; break; } // ب) اعتبارسنجی خروجی Hit if (!Number.isFinite(hit.t) || !Number.isFinite(hit.nx) || !Number.isFinite(hit.ny)) { throw new TypeError("castCircle returned an invalid hit object"); } const nLen = Math.hypot(hit.nx, hit.ny); if (nLen < 1e-12) { throw new TypeError("collision normal length must not be zero"); } const nx = hit.nx / nLen; const ny = hit.ny / nLen; const t = Math.max(0, Math.min(1, hit.t)); hitCount++; if (collectHits) { hits.push({ t, nx, ny }); } // ۴. پیشروی تا نقطه تماس با احتساب Skin در راستای نرمال if (t > 0) { entity.x += stepDx * t; entity.y += stepDy * t; } // اعمال پوسته‌ی اطمینان در راستای عمود بر سطح (جلوگیری از تغییر راستای مماسی) if (skin > 0) { entity.x += nx * skin; entity.y += ny * skin; } // کسر دقیق زمان مصرف‌شده timeLeft -= timeLeft * t; if (timeLeft <= 1e-4) { break; } // ۵. ثبت نرمال جدید سطح let duplicate = false; for (let i = 0; i < numPlanes; i++) { if (planesX[i] * nx + planesY[i] * ny > 0.999) { duplicate = true; break; } } if (!duplicate && numPlanes < maxIters) { planesX[numPlanes] = nx; planesY[numPlanes] = ny; numPlanes++; } // ۶. برش و اصلاح بردار سرعت (Clip Velocity) clipVelocity(nx, ny, restitution, friction); // ۷. کنترل کنج‌ها و لغزش بین دو دیوار (Corner/Crease Handling در فضای ۲D) let trapped = false; for (let i = 0; i < numPlanes; i++) { const dot = curVx * planesX[i] + curVy * planesY[i]; // اگر بردار به سمت پشت یکی از صفحات لغزش قبلی باشد: if (dot < -1e-5) { // برش مجدد روی صفحه قبلی clipVelocity(planesX[i], planesY[i], 0, friction); // اگر باز هم با صفحه فعلی تداخل داشت یعنی داخل کنج بسته گیر کرده است if (curVx * nx + curVy * ny < -1e-5) { trapped = true; break; } } } if (trapped) { curVx = 0; curVy = 0; break; } } // تابع درونی برش سرعت بر مبنای ضرب داخلی و نگاشت مماس function clipVelocity(normalX, normalY, bounce, fric) { const into = curVx * normalX + curVy * normalY; if (into < 0) { const normalCompX = into * normalX; const normalCompY = into * normalY; // مؤلفه مماس با احتساب اصطکاک const tangX = (curVx - normalCompX) * (1 - fric); const tangY = (curVy - normalCompY) * (1 - fric); // مؤلفه نرمال با ضریب بازگشت الاستیک curVx = tangX - normalCompX * bounce; curVy = tangY - normalCompY * bounce; } } return { x: entity.x, y: entity.y, vx: curVx, vy: curVy, movedX: entity.x - startX, movedY: entity.y - startY, hitCount, iterations, timeLeft, trapped: curVx === 0 && curVy === 0 && timeLeft > 0, ...(collectHits ? { collisions: hits } : {}), }; } function clamp01(value) { return value < 0 ? 0 : value > 1 ? 1 : value; }
